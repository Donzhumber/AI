// Extractor flexible de métricas numéricas
getMetrica = function(d, key) {
  if (!d || !key) return null;
  if (d[key] !== undefined && d[key] !== null) {
    const v = Number(d[key]);
    return isNaN(v) ? null : v;
  }
  const alt = key.startsWith("exp_") ? key.replace("exp_", "") : "exp_" + key;
  if (d[alt] !== undefined && d[alt] !== null) {
    const v = Number(d[alt]);
    return isNaN(v) ? null : v;
  }
  return null;
}

// Extractor de tasa de informalidad en porcentaje (0 - 100)
getInformalidad = function(d) {
  if (!d) return 0;
  let inf = d.tasa_informalidad ?? d.informalidad_pct ?? d.pct_informalidad ?? d.informalidad ?? d.tasa_informal ?? 0;

}

// Extractor de salario formateado a precios constantes de octubre de 2026
getSalarioTexto = function(d) {
  if (!d) return "N/D";

  const s = d.salario_medio ?? d.salario_promedio ?? d.salario;

}

// Incorpora cálculo automático de Media Ponderada y Mediana Ponderada Nacional
calcularMetricasCuadrantes = function(data, keyX, keyY, criterioUmbral = "media", customUmbralX = null, customUmbralY = null) {
  const validos = (data || []).filter(d => {
    const x = getMetrica(d, keyX);
    const y = getMetrica(d, keyY);
    return x !== null && !isNaN(x) && y !== null && !isNaN(y);
  });
  
  const totalPob = d3.sum(validos, d => d.poblacion_estimada || 0);

  // 1. Cálculo de Media Ponderada Nacional
  const wMeanX = totalPob > 0 ? d3.sum(validos, d => (getMetrica(d, keyX) || 0) * (d.poblacion_estimada || 0)) / totalPob : 0.5;
  const wMeanY = totalPob > 0 ? d3.sum(validos, d => (getMetrica(d, keyY) || 0) * (d.poblacion_estimada || 0)) / totalPob : 0.5;

  // 2. Cálculo de Mediana Ponderada Nacional
  const calcMedianaPond = (key) => {
    const ordenados = [...validos].sort((a, b) => (getMetrica(a, key) || 0) - (getMetrica(b, key) || 0));
    let acum = 0;
    for (const d of ordenados) {
      acum += (d.poblacion_estimada || 0) / (totalPob || 1);
      if (acum >= 0.5) return getMetrica(d, key);
    }
    return wMeanX;
  };
  const wMedianX = calcMedianaPond(keyX);
  const wMedianY = calcMedianaPond(keyY);

  // 3. Determinación de umbrales según criterio
  let umbralX = wMeanX;
  let umbralY = wMeanY;
  if (criterioUmbral === "mediana") {
    umbralX = wMedianX;
    umbralY = wMedianY;
  } else if (criterioUmbral === "fijo") {
    umbralX = umbralFijoPara(keyX);
    umbralY = umbralFijoPara(keyY);
  } else if (customUmbralX !== null && customUmbralY !== null) {
    umbralX = customUmbralX;
    umbralY = customUmbralY;
  }

  const cuadrantes = {
    Q1: { id: "Q1", nombre: "Alta Complementariedad / Bajo Riesgo", subtitulo: "Oportunidad y Potenciación", color: "#059669", bg: "#ecfdf5", border: "#a7f3d0", pob: 0, infSum: 0, count: 0 },
    Q2: { id: "Q2", nombre: "Alta Complementariedad / Alto Riesgo", subtitulo: "Reestructuración Acelerada", color: "#d97706", bg: "#fffbeb", border: "#fde68a", pob: 0, infSum: 0, count: 0 },
    Q3: { id: "Q3", nombre: "Baja Complementariedad / Bajo Riesgo", subtitulo: "Aislamiento Tecnológico", color: "#475569", bg: "#f8fafc", border: "#cbd5e1", pob: 0, infSum: 0, count: 0 },
    Q4: { id: "Q4", nombre: "Baja Complementariedad / Alto Riesgo", subtitulo: "Sustitución Directa / Automatización", color: "#dc2626", bg: "#fef2f2", border: "#fecaca", pob: 0, infSum: 0, count: 0 }
  };

  const procesados = validos.map(d => {
    const x = getMetrica(d, keyX);
    const y = getMetrica(d, keyY);
    const p = d.poblacion_estimada || 0;
    const inf = getInformalidad(d);

    let q = "Q3";
    if (x < umbralX && y >= umbralY) q = "Q1";
    else if (x >= umbralX && y >= umbralY) q = "Q2";
    else if (x < umbralX && y < umbralY) q = "Q3";
    else if (x >= umbralX && y < umbralY) q = "Q4";

    cuadrantes[q].pob += p;
    cuadrantes[q].infSum += (inf * p);
    cuadrantes[q].count += 1;

    return { ...d, _x: x, _y: y, _q: q };
  });

  for (const q of Object.keys(cuadrantes)) {
    const c = cuadrantes[q];
    c.pctOcupados = totalPob > 0 ? (c.pob / totalPob) * 100 : 0;
    c.pctInformalidad = c.pob > 0 ? (c.infSum / c.pob) : 0;
  }

  return { 
    cuadrantes, 
    totalPob, 
    totalOcupaciones: validos.length, 
    validos: procesados,
    umbralX,
    umbralY,
    wMeanX,
    wMeanY,
    wMedianX,
    wMedianY
  };
}

// Mini-pirámide poblacional (SVG de Observable Plot, resuelto de forma
// síncrona a string) para el popover de hover de cada tarjeta de cuadrante.
miniPiramideSVG = function(datos) {
  if (!datos || datos.length === 0) {
    return `<div style="font-size:0.7rem;color:#94a3b8;padding:16px;text-align:center;">Sin datos suficientes</div>`;
  }
  const totalCuadrante = d3.sum(datos, d => d.poblacion) || 1;

  // Replica el ancho real de trazado (width - marginLeft - marginRight) para
  // saber cuántos px ocupa cada barra y decidir si el % en blanco cabe adentro.
  const anchoTrazadoPx = 220 - 42 - 6; // 172
  const valoresX = datos.map(d => d.sexo === "Hombre" ? -d.poblacion : d.poblacion);
  const domMin = Math.min(0, ...valoresX);
  const domMax = Math.max(0, ...valoresX);
  const pxPorUnidad = anchoTrazadoPx / ((domMax - domMin) || 1);
  const anchoBarraPx = d => Math.abs(d.poblacion) * pxPorUnidad;
  const UMBRAL_PX = 18; // ancho mínimo de barra para que el % en blanco quepa/legible

  const plot = Plot.plot({
    width: 220, height: 190, marginLeft: 42, marginRight: 6, marginTop: 4, marginBottom: 18,
    x: { axis: null },
    y: { domain: ordenEdades10, label: null },
    color: { domain: ["Hombre", "Mujer"], range: ["#2b5c8f", "#d95f02"], legend: true },
    marks: [
      Plot.barX(datos, { y: "edad_grupo10", x: d => d.sexo === "Hombre" ? -d.poblacion : d.poblacion, fill: "sexo" }),
      Plot.ruleX([0], { stroke: "#333", strokeWidth: 1 }),
      // % sobre el total del cuadrante, centrado dentro de la barra (x = mitad
      // de camino entre 0 y el extremo; offset horneado en x, no dx/dy).
      Plot.text(
        datos.filter(d => anchoBarraPx(d) >= UMBRAL_PX),
        {
          y: "edad_grupo10",
          x: d => (d.sexo === "Hombre" ? -d.poblacion : d.poblacion) / 2,
          text: d => `${(100 * d.poblacion / totalCuadrante).toFixed(0)}%`,
          fill: "#ffffff",
          fontSize: 7,
          fontWeight: "700",
          textAnchor: "middle"
        }
      )
    ]
  });
  const html = plot.outerHTML;
  plot.remove();
  return html;
}
